```bash
python - <<'PY'
from pathlib import Path
import os, shutil
src_dirs = [
    Path('data/images/train/0'),
    Path('data/images_supervised/train'),
    Path('data/images_supervised/val'),
    Path('data/images_supervised/test'),
]
dst = Path('data/images_all/0')
dst.mkdir(parents=True, exist_ok=True)
for s in src_dirs:
    if not s.exists():
        continue
    for p in s.rglob('*'):
        if p.is_file() and p.suffix.lower() in {'.jpg', '.jpeg', '.png', '.bmp', '.tif', '.tiff'}:
            t = dst / p.name
            if t.exists():
                continue
            try:
                os.symlink(p.resolve(), t)
            except Exception:
                shutil.copy2(p, t)
print('images_all ready:', len(list(dst.glob('*'))))
PY
```

```bash
for p in 20 50 80 100; do
  mkdir -p data/ssl_subsets/${p}/train/0
  P=$p python - <<'PY'
import os, pandas as pd, shutil
from pathlib import Path
p = os.environ['P']
df = pd.read_csv(f'data/splits_proportions/{p}%/train.csv')
dst = Path(f'data/ssl_subsets/{p}/train/0')
roots = [Path('data/images_all/0'), Path('data/images/train/0'), Path('data/images_supervised/train')]
for fp in df['filepath'].astype(str):
    name = Path(fp).name
    src = None
    for r in roots:
        c = r / name
        if c.exists():
            src = c
            break
    if src is None:
        continue
    t = dst / name
    if t.exists():
        continue
    try:
        os.symlink(src.resolve(), t)
    except Exception:
        shutil.copy2(src, t)
print(f'[OK] subset {p}%')
PY
done
```

## 4) TAPT（20/50/80/100）四并发命令

```bash
for p in 20 50 80 100; do
  CUDA_VISIBLE_DEVICES=0 nohup python core/train_ssl.py \
    --config-file configs/pretrain/vits16_domain_adapt.yaml \
    --output-dir ./outputs/tapt_vits16_${p} \
    train.dataset_path=./data/ssl_subsets/${p}/train \
    student.pretrained_weights=./model/dinov3_vits16_pretrain.pth \
    student.arch=vit_small \
    > ./logs/tapt_vits16_${p}.log 2>&1 &

  CUDA_VISIBLE_DEVICES=1 nohup python core/train_ssl.py \
    --config-file configs/pretrain/vits16plus_domain_adapt.yaml \
    --output-dir ./outputs/tapt_vits16plus_${p} \
    train.dataset_path=./data/ssl_subsets/${p}/train \
    student.pretrained_weights=./model/dinov3_vits16_pretrain.pth \
    student.arch=vit_small \
    > ./logs/tapt_vits16plus_${p}.log 2>&1 &

  CUDA_VISIBLE_DEVICES=2 nohup python core/train_ssl.py \
    --config-file configs/pretrain/vitb16_domain_adapt.yaml \
    --output-dir ./outputs/tapt_vitb16_${p} \
    train.dataset_path=./data/ssl_subsets/${p}/train \
    student.pretrained_weights=./model/dinov3_vitb16_pretrain.pth \
    student.arch=vit_base \
    > ./logs/tapt_vitb16_${p}.log 2>&1 &

  CUDA_VISIBLE_DEVICES=3 nohup python core/train_ssl.py \
    --config-file configs/pretrain/vitl16_domain_adapt.yaml \
    --output-dir ./outputs/tapt_vitl16_${p} \
    train.dataset_path=./data/ssl_subsets/${p}/train \
    student.pretrained_weights=./model/dinov3_vitl16_pretrain.pth \
    student.arch=vit_large \
    > ./logs/tapt_vitl16_${p}.log 2>&1 &

  wait
  echo "[DONE] ${p}%"
done
```
